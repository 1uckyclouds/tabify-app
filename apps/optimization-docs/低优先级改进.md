```typescript
  selectedTabs: string[];

  tabs,
  selectedTabs,
  onSelectTab,
  containerHeight
}) => {
  const parentRef = React.useRef<HTMLDivElement>(null);

  const virtualizer = useVirtualizer({
    count: tabs.length,
    getScrollElement: () => parentRef.current,
    estimateSize: () => 120, // 预估每个标签页项目高度

          width: '100%',
          position: 'relative',
        }}
      >

          const tab = tabs[virtualItem.index];
          return (
            <div
              key={virtualItem.key}
              style={{
                position: 'absolute',
                top: 0,
                left: 0,
                width: '100%',

                transform: `translateY(${virtualItem.start}px)`,

              <div className="p-2">
                <TabItem
                  tab={tab}
                  isSelected={selectedTabs.includes(tab.id)}
                  onSelect={() => onSelectTab(tab.id)}
                />
              </div>
            </div>
          );
        })}
      </div>
    </div>
  );
```

#### 1.2 图片懒加载和缓存
```typescript
// hooks/useImageCache.ts
import { useState, useEffect } from 'react';

interface ImageCacheEntry {
  url: string;
  blob: Blob;
  timestamp: number;
}

class ImageCache {
  private cache = new Map<string, ImageCacheEntry>();
  private maxSize = 100; // 最大缓存数量
  private maxAge = 24 * 60 * 60 * 1000; // 24小时过期

  async get(url: string): Promise<string | null> {
    const entry = this.cache.get(url);
    
    if (entry && Date.now() - entry.timestamp < this.maxAge) {
      return URL.createObjectURL(entry.blob);
    }
    
    try {
      const response = await fetch(url);
      const blob = await response.blob();
      
      // 清理过期缓存
      this.cleanup();
      
      // 添加到缓存
      this.cache.set(url, {
        url,
        blob,
        timestamp: Date.now()
      });
      
      return URL.createObjectURL(blob);
    } catch (error) {
      console.warn('Failed to cache image:', url, error);
      return null;
    }
  }

  private cleanup() {
    if (this.cache.size >= this.maxSize) {
      const entries = Array.from(this.cache.entries());
      entries.sort((a, b) => a[1].timestamp - b[1].timestamp);
      
      // 删除最旧的一半
      const toDelete = entries.slice(0, Math.floor(entries.length / 2));
      toDelete.forEach(([key]) => this.cache.delete(key));
    }
  }
}

const imageCache = new ImageCache();

export const useImageCache = (url: string) => {
  const [cachedUrl, setCachedUrl] = useState<string | null>(null);
  const [loading, setLoading] = useState(true);
  const [error, setError] = useState(false);

  useEffect(() => {
    if (!url) {
      setLoading(false);
      return;
    }

    setLoading(true);
    setError(false);

    imageCache.get(url)
      .then(cached => {
        setCachedUrl(cached || url);
        setLoading(false);
      })
      .catch(() => {
        setError(true);
        setLoading(false);
        setCachedUrl(url); // 回退到原始URL
      });
  }, [url]);

  return { cachedUrl, loading, error };
};
```

#### 1.3 防抖搜索优化
```typescript
// hooks/useDebouncedSearch.ts
import { useState, useEffect, useMemo } from 'react';
import { Tab } from '../types';

interface SearchOptions {
  debounceMs?: number;
  minQueryLength?: number;
  searchFields?: (keyof Tab)[];
}

export const useDebouncedSearch = (
  tabs: Tab[],
  query: string,
  options: SearchOptions = {}
) => {
  const {
    debounceMs = 300,
    minQueryLength = 2,
    searchFields = ['title', 'url']
  } = options;

  const [debouncedQuery, setDebouncedQuery] = useState(query);
  const [isSearching, setIsSearching] = useState(false);

  // 防抖处理
  useEffect(() => {
    setIsSearching(true);
    const timer = setTimeout(() => {
      setDebouncedQuery(query);
      setIsSearching(false);
    }, debounceMs);

    return () => clearTimeout(timer);
  }, [query, debounceMs]);

  // 搜索结果
  const searchResults = useMemo(() => {
    if (!debouncedQuery || debouncedQuery.length < minQueryLength) {
      return tabs;
    }

    const lowercaseQuery = debouncedQuery.toLowerCase();
    
    return tabs.filter(tab => 
      searchFields.some(field => {
        const value = tab[field];
        return typeof value === 'string' && 
               value.toLowerCase().includes(lowercaseQuery);
      })
    );
  }, [tabs, debouncedQuery, minQueryLength, searchFields]);

  return {
    searchResults,
    isSearching: isSearching && query.length >= minQueryLength,
    hasQuery: debouncedQuery.length >= minQueryLength
  };
};
```

```typescript
export const ThemeProvider: React.FC<{ children: React.ReactNode }> = ({ children }) => {
  const [currentTheme, setCurrentTheme] = useState<Theme>(themes[0]);

  useEffect(() => {
    const saved = localStorage.getItem('tabify-theme');
    if (saved) {
      const savedTheme = JSON.parse(saved);
      const existingTheme = themes.find(t => t.id === savedTheme.id);
      setCurrentTheme(existingTheme || savedTheme);
    }
  }, []);

  const setTheme = (themeId: string) => {
    const theme = themes.find(t => t.id === themeId);
    if (theme) {
      setCurrentTheme(theme);
      localStorage.setItem('tabify-theme', JSON.stringify(theme));
    }
  };

  const customizeTheme = (updates: Partial<Theme>) => {
    const customTheme = { ...currentTheme, ...updates };
    setCurrentTheme(customTheme);
    localStorage.setItem('tabify-theme', JSON.stringify(customTheme));
  };

  // 应用CSS变量
  useEffect(() => {
    const root = document.documentElement;
    Object.entries(currentTheme.colors).forEach(([key, value]) => {
      root.style.setProperty(`--color-${key}`, value);
    });
    root.style.setProperty('--border-radius', 
      currentTheme.borderRadius === 'none' ? '0' :
      currentTheme.borderRadius === 'sm' ? '0.25rem' :
      currentTheme.borderRadius === 'md' ? '0.375rem' :
      currentTheme.borderRadius === 'lg' ? '0.5rem' : '0.75rem'
    );
  }, [currentTheme]);

  return (
    <ThemeContext.Provider value={{
      currentTheme,
      themes,
      setTheme,
      customizeTheme
    }}>
      {children}
    </ThemeContext.Provider>
  );
};

export const useTheme = () => {
  const context = useContext(ThemeContext);
  if (!context) {
    throw new Error('useTheme must be used within ThemeProvider');
  }
  return context;
};
```

#### 3.1 使用统计Hook
```typescript
// hooks/useAnalytics.ts
import { useState, useEffect } from 'react';

interface AnalyticsData {
  totalTabs: number;
  totalGroups: number;
  mostVisitedDomains: { domain: string; count: number }[];
  dailyActivity: { date: string; tabsOpened: number; tabsClosed: number }[];
  averageTabsPerDay: number;
  topCategories: { category: string; count: number }[];
}

interface UsageEvent {
  type: 'tab_opened' | 'tab_closed' | 'tab_grouped' | 'search_performed';
  timestamp: number;
  data: any;
}

export const useAnalytics = () => {
  const [analytics, setAnalytics] = useState<AnalyticsData | null>(null);
  const [events, setEvents] = useState<UsageEvent[]>([]);

  // 加载历史数据
  useEffect(() => {
    const savedEvents = localStorage.getItem('tabify-analytics');
    if (savedEvents) {
      setEvents(JSON.parse(savedEvents));
    }
  }, []);

  // 记录事件
  const trackEvent = (type: UsageEvent['type'], data: any) => {
    const event: UsageEvent = {
      type,
      timestamp: Date.now(),
      data
    };
    
    const newEvents = [...events, event];
    setEvents(newEvents);
    localStorage.setItem('tabify-analytics', JSON.stringify(newEvents));
  };

  // 计算统计数据
  useEffect(() => {
    if (events.length === 0) return;

    const now = Date.now();
    const thirtyDaysAgo = now - (30 * 24 * 60 * 60 * 1000);
    const recentEvents = events.filter(e => e.timestamp > thirtyDaysAgo);

    // 域名统计
    const domainCounts = new Map<string, number>();
    recentEvents
      .filter(e => e.type === 'tab_opened' && e.data.url)
      .forEach(e => {
        try {
          const domain = new URL(e.data.url).hostname;
          domainCounts.set(domain, (domainCounts.get(domain) || 0) + 1);
        } catch {}
      });

    const mostVisitedDomains = Array.from(domainCounts.entries())
      .map(([domain, count]) => ({ domain, count }))
      .sort((a, b) => b.count - a.count)
      .slice(0, 10);

    // 每日活动统计
    const dailyActivity = new Map<string, { tabsOpened: number; tabsClosed: number }>();
    recentEvents.forEach(e => {
      const date = new Date(e.timestamp).toISOString().split('T')[0];
      const activity = dailyActivity.get(date) || { tabsOpened: 0, tabsClosed: 0 };
      
      if (e.type === 'tab_opened') activity.tabsOpened++;
      if (e.type === 'tab_closed') activity.tabsClosed++;
      
      dailyActivity.set(date, activity);
    });

    const dailyActivityArray = Array.from(dailyActivity.entries())
      .map(([date, data]) => ({ date, ...data }))
      .sort((a, b) => a.date.localeCompare(b.date));

    setAnalytics({
      totalTabs: recentEvents.filter(e => e.type === 'tab_opened').length,
      totalGroups: recentEvents.filter(e => e.type === 'tab_grouped').length,
      mostVisitedDomains,
      dailyActivity: dailyActivityArray,
      averageTabsPerDay: dailyActivityArray.length > 0 
        ? dailyActivityArray.reduce((sum, day) => sum + day.tabsOpened, 0) / dailyActivityArray.length
        : 0,
      topCategories: [] // 可以根据域名分类
    });
  }, [events]);

  return {
    analytics,
    trackEvent
  };
};
```

```typescript
                  return (
                    <div key={day.date} className="flex-1 flex flex-col items-center">
                      <div

                        style={{ height: `${height}%`, minHeight: height > 0 ? '4px' : '0' }}

                      <span className="text-xs text-gray-500 mt-1 transform rotate-45 origin-left">
                        {new Date(day.date).getDate()}
                      </span>
```

```typescript
// hooks/useSnapshots.ts
import { useState, useEffect } from 'react';
import { Tab, TabGroup } from '../types';

interface Snapshot {
  id: string;
  name: string;
  timestamp: number;
  tabs: Tab[];
  groups: TabGroup[];
  description?: string;
}

export const useSnapshots = () => {
  const [snapshots, setSnapshots] = useState<Snapshot[]>([]);

  useEffect(() => {
    const saved = localStorage.getItem('tabify-snapshots');
    if (saved) {
      setSnapshots(JSON.parse(saved));
    }
  }, []);

  const saveSnapshot = (name: string, tabs: Tab[], groups: TabGroup[], description?: string) => {
    const snapshot: Snapshot = {
      id: Date.now().toString(),
      name,
      timestamp: Date.now(),
      tabs: [...tabs],
      groups: [...groups],
      description
    };

    const newSnapshots = [snapshot, ...snapshots].slice(0, 50); // 最多保存50个快照
    setSnapshots(newSnapshots);
    localStorage.setItem('tabify-snapshots', JSON.stringify(newSnapshots));
    
    return snapshot;
  };

  const deleteSnapshot = (id: string) => {
    const newSnapshots = snapshots.filter(s => s.id !== id);
    setSnapshots(newSnapshots);
    localStorage.setItem('tabify-snapshots', JSON.stringify(newSnapshots));
  };

  const restoreSnapshot = (id: string) => {
    const snapshot = snapshots.find(s => s.id === id);
    return snapshot ? { tabs: snapshot.tabs, groups: snapshot.groups } : null;
  };

  return {
    snapshots,
    saveSnapshot,
    deleteSnapshot,
    restoreSnapshot
  };
};
```

#### 4.2 智能分组算法
```typescript
// utils/smartGrouping.ts
import { Tab } from '../types';

interface GroupingRule {
  type: 'domain' | 'keyword' | 'time' | 'similarity';
  weight: number;
}

class SmartGrouping {
  private rules: GroupingRule[] = [
    { type: 'domain', weight: 0.4 },
    { type: 'keyword', weight: 0.3 },
    { type: 'similarity', weight: 0.2 },
    { type: 'time', weight: 0.1 }
  ];

  groupTabs(tabs: Tab[]): { [groupName: string]: Tab[] } {
    const groups: { [key: string]: Tab[] } = {};
    
    // 按域名分组
    const domainGroups = this.groupByDomain(tabs);
    
    // 按关键词分组
    const keywordGroups = this.groupByKeywords(tabs);
    
    // 合并分组结果
    Object.entries(domainGroups).forEach(([domain, domainTabs]) => {
      if (domainTabs.length >= 3) { // 只有3个以上标签页才创建域名分组
        groups[`${domain} 相关`] = domainTabs;
      } else {
        // 尝试按关键词分组
        domainTabs.forEach(tab => {
          const keywordGroup = this.findBestKeywordGroup(tab, keywordGroups);
          if (keywordGroup) {
            if (!groups[keywordGroup]) groups[keywordGroup] = [];
            groups[keywordGroup].push(tab);
          } else {
            if (!groups['其他']) groups['其他'] = [];
            groups['其他'].push(tab);
          }
        });
      }
    });

    return groups;
  }

  private groupByDomain(tabs: Tab[]): { [domain: string]: Tab[] } {
    const groups: { [domain: string]: Tab[] } = {};
    
    tabs.forEach(tab => {
      try {
        const domain = new URL(tab.url).hostname;
        if (!groups[domain]) groups[domain] = [];
        groups[domain].push(tab);
      } catch {
        if (!groups['invalid']) groups['invalid'] = [];
        groups['invalid'].push(tab);
      }
    });
    
    return groups;
  }

  private groupByKeywords(tabs: Tab[]): { [keyword: string]: Tab[] } {
    const keywords = ['github', 'stackoverflow', 'documentation', 'tutorial', 'api', 'react', 'vue', 'angular'];
    const groups: { [keyword: string]: Tab[] } = {};
    
    tabs.forEach(tab => {
      const title = tab.title.toLowerCase();
      const url = tab.url.toLowerCase();
      
      keywords.forEach(keyword => {
        if (title.includes(keyword) || url.includes(keyword)) {
          if (!groups[keyword]) groups[keyword] = [];
          groups[keyword].push(tab);
        }
      });
    });
    
    return groups;
  }

  private findBestKeywordGroup(tab: Tab, keywordGroups: { [keyword: string]: Tab[] }): string | null {
    const title = tab.title.toLowerCase();
    const url = tab.url.toLowerCase();
    
    for (const [keyword, groupTabs] of Object.entries(keywordGroups)) {
      if (groupTabs.includes(tab)) {
        return `${keyword} 相关`;
      }
    }
    
    return null;
  }
}

export const smartGrouping = new SmartGrouping();
```

```typescript
// utils/importExport.ts
import { Tab, TabGroup } from '../types';

interface ExportData {
  version: string;
  timestamp: number;
  tabs: Tab[];
  groups: TabGroup[];
  metadata: {
    totalTabs: number;
    totalGroups: number;
    exportedBy: string;
  };
}

export class ImportExport {
  static export(tabs: Tab[], groups: TabGroup[]): string {
    const data: ExportData = {
      version: '1.0.0',
      timestamp: Date.now(),
      tabs,
      groups,
      metadata: {
        totalTabs: tabs.length,
        totalGroups: groups.length,
        exportedBy: 'Tabify'
      }
    };
    
    return JSON.stringify(data, null, 2);
  }

  static import(jsonString: string): { tabs: Tab[]; groups: TabGroup[] } | null {
    try {
      const data: ExportData = JSON.parse(jsonString);
      
      // 验证数据格式
      if (!data.version || !data.tabs || !data.groups) {
        throw new Error('Invalid data format');
      }
      
      return {
        tabs: data.tabs,
        groups: data.groups
      };
    } catch (error) {
      console.error('Import failed:', error);
      return null;
    }
  }

  static exportToFile(tabs: Tab[], groups: TabGroup[], filename?: string) {
    const data = this.export(tabs, groups);
    const blob = new Blob([data], { type: 'application/json' });
    const url = URL.createObjectURL(blob);
    
    const a = document.createElement('a');
    a.href = url;
    a.download = filename || `tabify-export-${new Date().toISOString().split('T')[0]}.json`;
    document.body.appendChild(a);
    a.click();
    document.body.removeChild(a);
    URL.revokeObjectURL(url);
  }

  static importFromFile(): Promise<{ tabs: Tab[]; groups: TabGroup[] } | null> {
    return new Promise((resolve) => {
      const input = document.createElement('input');
      input.type = 'file';
      input.accept = '.json';
      
      input.onchange = (e) => {
        const file = (e.target as HTMLInputElement).files?.[0];
        if (!file) {
          resolve(null);
          return;
        }
        
        const reader = new FileReader();
        reader.onload = (e) => {
          const content = e.target?.result as string;
          const result = this.import(content);
          resolve(result);
        };
        reader.readAsText(file);
      };
      
      input.click();
    });
  }
}
```
